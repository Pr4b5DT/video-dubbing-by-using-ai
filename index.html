# Python FastAPI Backend Implementation
import os, subprocess
from fastapi import FastAPI, BackgroundTasks
from openai import OpenAI
from elevenlabs.client import ElevenLabs

app = FastAPI()

@app.post("/api/dub-video")
async def dub_video(youtube_url: str, target_lang: str):
    # 1. Download YouTube Audio via yt-dlp
    cmd_download = f'yt-dlp -x --audio-format mp3 -o "temp_audio.mp3" {youtube_url}'
    subprocess.run(cmd_download, shell=True)

    # 2. Transcribe Audio via OpenAI Whisper
    client = OpenAI()
    with open("temp_audio.mp3", "rb") as audio_file:
        transcript = client.audio.transcriptions.create(model="whisper-1", file=audio_file)

    # 3. Translate Text using GPT-4o
    response = client.chat.completions.create(
        model="gpt-4o",
        messages=[{"role": "user", "content": f"Translate to {target_lang}: {transcript.text}"}]
    )
    translated_text = response.choices[0].message.content

    # 4. Generate AI Voice via ElevenLabs
    eleven = ElevenLabs(api_key=os.getenv("ELEVENLABS_API_KEY"))
    audio = eleven.generate(text=translated_text, voice="Rachel", model="eleven_multilingual_v2")

    # 5. Merge Audio with original Video using FFmpeg
    subprocess.run("ffmpeg -i temp_video.mp4 -i dubbed.mp3 -c:v copy -c:a aac output.mp4", shell=True)
    return {"status": "success", "download_url": "/downloads/output.mp4"}
